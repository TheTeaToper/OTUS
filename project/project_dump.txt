// Вспомогательная функция отправки запросов на сервер
func sendRequest(method, url string, body []byte, successMsg string) {
	req, err := http.NewRequest(method, url, bytes.NewBuffer(body))
	if err != nil {
		fmt.Printf("❌ Ошибка создания запроса: %v\n", err)
		return
	}
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		fmt.Printf("❌ Сервер недоступен: %v. Сначала запустите основной сервис.\n", err)
		return
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusOK {
		fmt.Printf("✅ %s\n", successMsg)
	} else {
		type errResp struct{ Error string }
		var e errResp
		_ = json.NewDecoder(resp.Body).Decode(&e)
		if e.Error != "" {
			fmt.Printf("❌ Ошибка сервера (%d): %s\n", resp.StatusCode, e.Error)
		} else {
			fmt.Printf("❌ Сервер вернул ошибку со статусом: %d\n", resp.StatusCode)
		}
	}
}

package main

import (
	"context"
	"errors"
	"log"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	httpserver "github.com/TheTeaToper/OTUS/project/internal/server/http"

	"github.com/TheTeaToper/OTUS/project/internal/app"
	"github.com/TheTeaToper/OTUS/project/internal/config"
)

func main() {
	log.Println("Инициализация сервиса Анти-брутфорс...")

	// 1. Загружаем конфигурацию
	configPath := "configs/config.yaml"
	cfg, err := config.Load(configPath)
	if err != nil {
		log.Fatalf("Ошибка загрузки конфигурации: %v", err)
	}
	log.Printf("Конфигурация успешно загружена из %s", configPath)

	// 2. Инициализируем бизнес-логику (сервис)
	appConfig := app.Config{
		LoginLimit:    cfg.Limits.LoginLimit,
		PasswordLimit: cfg.Limits.PasswordLimit,
		IpLimit:       cfg.Limits.IpLimit,
	}
	service := app.NewAntibruteforceService(appConfig, cfg.Limits.BucketTTL)

	// 3. Создаем HTTP сервер
	handler := httpserver.NewServer(service)
	server := &http.Server{
		Addr:         cfg.Server.Port,
		Handler:      handler,
		ReadTimeout:  5 * time.Second,
		WriteTimeout: 10 * time.Second,
	}

	// Канал для перехвата системных сигналов остановки (Ctrl+C, kill)
	stop := make(chan os.Signal, 1)
	signal.Notify(stop, os.Interrupt, syscall.SIGTERM)

	// 4. Запускаем сервер в отдельной горутине
	go func() {
		log.Printf("HTTP сервер запущен на порту %s", cfg.Server.Port)
		if err := server.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			log.Fatalf("Ошибка работы HTTP сервера: %v", err)
		}
	}()

	// Ждем сигнал остановки
	<-stop
	log.Println("Получен сигнал остановки. Завершение работы...")

	// Плавное завершение работы (даем серверу 5 секунд на закрытие текущих соединений)
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	if err := server.Shutdown(ctx); err != nil {
		log.Fatalf("Ошибка при плавном завершении работы сервера: %v", err)
	}

	log.Println("Сервис Анти-брутфорс успешно остановлен.")
}

package app

import (
	"time"

	"github.com/TheTeaToper/OTUS/project/internal/limiter"
	"github.com/TheTeaToper/OTUS/project/internal/subnets"
)

type Config struct {
	// N
	LoginLimit int
	// M
	PasswordLimit int
	// K
	IpLimit int
}

type AntibruteforceService struct {
	config           Config
	blackList        *subnets.Subnets
	whiteList        *subnets.Subnets
	loginsManager    *limiter.Manager
	passwordsManager *limiter.Manager
	ipsManager       *limiter.Manager
}

func NewAntibruteforceService(cfg Config, bucketTtl time.Duration) *AntibruteforceService {
	return &AntibruteforceService{
		config:           cfg,
		blackList:        subnets.NewSubnets(),
		whiteList:        subnets.NewSubnets(),
		loginsManager:    limiter.NewManager(bucketTtl),
		passwordsManager: limiter.NewManager(bucketTtl),
		ipsManager:       limiter.NewManager(bucketTtl),
	}
}

func (s *AntibruteforceService) ValidateAuthorization(login, password, ip string) bool {
	// Проверка blackSubnets в приоритете
	if s.blackList.Contains(ip) {
		return false
	}
	if s.whiteList.Contains(ip) {
		return true
	}

	loginBucket := s.loginsManager.GetBucket(login, s.config.LoginLimit)
	passwordBucket := s.passwordsManager.GetBucket(password, s.config.PasswordLimit)
	ipBucket := s.ipsManager.GetBucket(ip, s.config.IpLimit)

	loginAllowed := loginBucket.Allow()
	passwordAllowed := passwordBucket.Allow()
	ipAllowed := ipBucket.Allow()

	return loginAllowed && passwordAllowed && ipAllowed
}

// Сброс бакетов для логина и IP
func (s *AntibruteforceService) ResetBuckets(login, ip string) {
	s.loginsManager.ResetKey(login)
	s.ipsManager.ResetKey(ip)
}

// Методы-прослойки для управления списками (вызываются из API/CLI)
func (s *AntibruteforceService) AddToWhitelist(cidr string) error { return s.whiteList.Add(cidr) }

func (s *AntibruteforceService) RemoveFromWhitelist(cidr string) error {
	return s.whiteList.Delete(cidr)
}
func (s *AntibruteforceService) AddToBlacklist(cidr string) error { return s.blackList.Add(cidr) }
func (s *AntibruteforceService) RemoveFromBlacklist(cidr string) error {
	return s.blackList.Delete(cidr)
}

package limiter

import (
	"sync"
	"time"
)

type Bucket struct {
	mutex sync.Mutex
	// Сколько токенов добавляется в секунду (Лимит / 60)
	rate float64
	// Максимальное количество токенов (равно N, M или K)
	capacity float64
	// Текущее количество доступных токенов
	tokens float64
	// Время последнего обновления токенов
	lastRefreshed time.Time
}

func NewBucket(limitPerMinute int) *Bucket {
	return &Bucket{
		rate:          float64(limitPerMinute) / 60.0,
		capacity:      float64(limitPerMinute),
		tokens:        float64(limitPerMinute),
		lastRefreshed: time.Now(),
	}
}

func (b *Bucket) Allow() bool {
	b.mutex.Lock()
	defer b.mutex.Unlock()

	now := time.Now()
	elapsed := now.Sub(b.lastRefreshed).Seconds()

	if elapsed > 0 {
		b.tokens += elapsed * b.rate
		if b.tokens > b.capacity {
			b.tokens = b.capacity
		}
		b.lastRefreshed = now
	}

	if b.tokens >= 1.0 {
		b.tokens -= 1.0
		return true
	}
	return false
}

func (b *Bucket) Reset() {
	b.mutex.Lock()
	defer b.mutex.Unlock()

	b.tokens = b.capacity
	b.lastRefreshed = time.Now()
}

package limiter

import (
	"sync"
	"time"
)

type Manager struct {
	mutex   sync.RWMutex
	buckets map[string]*Bucket
	// Время жизни неактивного бакета (например, 5-10 минут)
	ttl time.Duration
}

func NewManager(ttl time.Duration) *Manager {
	manager := &Manager{
		buckets: make(map[string]*Bucket),
		ttl:     ttl,
	}
	// Опционально: здесь можно запустить тикер для периодической очистки старых бакетов
	return manager
}

func (m *Manager) GetBucket(key string, limit int) *Bucket {
	m.mutex.RLock()
	bucket, exists := m.buckets[key]
	m.mutex.RUnlock()

	if exists {
		return bucket
	}

	m.mutex.Lock()
	defer m.mutex.Unlock()

	// Двойная проверка на случай race condition
	if bucket, exists = m.buckets[key]; exists {
		return bucket
	}

	bucket = NewBucket(limit)
	m.buckets[key] = bucket
	return bucket
}

// ResetKey удаляет/сбрасывает бакеты для конкретного ключа (нужно для метода Сброса)
func (m *Manager) ResetKey(key string) {
	m.mutex.Lock()
	defer m.mutex.Unlock()

	delete(m.buckets, key)
}

package config

import (
	"fmt"
	"os"
	"time"

	"gopkg.in/yaml.v3"
)

// Config объединяет все настройки приложения
type Config struct {
	Server struct {
		Port string `yaml:"port"`
	} `yaml:"server"`

	Limits struct {
		LoginLimit    int           `yaml:"login_limit"`    // N
		PasswordLimit int           `yaml:"password_limit"` // M
		IpLimit       int           `yaml:"ip_limit"`       // K
		BucketTTL     time.Duration `yaml:"bucket_ttl"`     // Время жизни неактивных бакетов
	} `yaml:"limits"`
}

// Load считывает конфигурацию из указанного файла
func Load(path string) (*Config, error) {
	cfg := &Config{}

	// Устанавливаем дефолтные значения на случай, если в файле что-то пропущено
	cfg.Server.Port = ":8080"
	cfg.Limits.LoginLimit = 10
	cfg.Limits.PasswordLimit = 100
	cfg.Limits.IpLimit = 1000
	cfg.Limits.BucketTTL = 10 * time.Minute

	// Проверяем наличие файла
	file, err := os.Open(path)
	if err != nil {
		return nil, fmt.Errorf("failed to open config file: %w", err)
	}
	defer file.Close()

	// Декодируем YAML
	decoder := yaml.NewDecoder(file)
	if err := decoder.Decode(cfg); err != nil {
		return nil, fmt.Errorf("failed to decode config YAML: %w", err)
	}

	return cfg, nil
}

package http

import (
	"encoding/json"
	"net/http"

	"github.com/TheTeaToper/OTUS/project/internal/app"
)

type Server struct {
	service *app.AntibruteforceService
	mux     *http.ServeMux
}

func NewServer(service *app.AntibruteforceService) *Server {
	server := &Server{
		service: service,
		mux:     http.NewServeMux(),
	}
	server.mux.HandleFunc("POST /api/v1/blacklist", server.handleAddToBlacklist)
	server.mux.HandleFunc("DELETE /api/v1/blacklist", server.handleRemoveFromBlacklist)

	server.mux.HandleFunc("POST /api/v1/whitelist", server.handleAddToWhitelist)
	server.mux.HandleFunc("DELETE /api/v1/whitelist", server.handleRemoveFromWhitelist)

	server.mux.HandleFunc("POST /api/v1/auth/validate", server.handleValidateAuthorization)
	server.mux.HandleFunc("POST /api/v1/buckets/reset", server.handleResetBuckets)

	return server
}

func (s *Server) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	s.mux.ServeHTTP(w, r)
}

func (s *Server) handleAddToBlacklist(w http.ResponseWriter, r *http.Request) {
	s.modifyList(w, r, s.service.AddToBlacklist)
}

func (s *Server) handleRemoveFromBlacklist(w http.ResponseWriter, r *http.Request) {
	s.modifyList(w, r, s.service.RemoveFromBlacklist)
}

func (s *Server) handleAddToWhitelist(w http.ResponseWriter, r *http.Request) {
	s.modifyList(w, r, s.service.AddToWhitelist)
}

func (s *Server) handleRemoveFromWhitelist(w http.ResponseWriter, r *http.Request) {
	s.modifyList(w, r, s.service.RemoveFromWhitelist)
}

func (s *Server) handleValidateAuthorization(w http.ResponseWriter, r *http.Request) {
	var request AuthRequest
	if err := json.NewDecoder(r.Body).Decode(&request); err != nil {
		s.respondWithError(w, http.StatusBadRequest, "invalid json body")
		return
	}

	if request.Login == "" || request.Password == "" || request.IP == "" {
		s.respondWithError(w, http.StatusBadRequest, "login, password and ip are required")
		return
	}

	ok := s.service.ValidateAuthorization(request.Login, request.Password, request.IP)
	s.respondWithJSON(w, http.StatusOK, AuthResponse{OK: ok})
}

func (s *Server) handleResetBuckets(w http.ResponseWriter, r *http.Request) {
	var req ResetRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		s.respondWithError(w, http.StatusBadRequest, "invalid json body")
		return
	}

	if req.Login == "" || req.IP == "" {
		s.respondWithError(w, http.StatusBadRequest, "login and ip are required")
		return
	}

	s.service.ResetBuckets(req.Login, req.IP)
	w.WriteHeader(http.StatusOK)
}

func (s *Server) modifyList(w http.ResponseWriter, r *http.Request, action func(string) error) {
	var req CidrRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		s.respondWithError(w, http.StatusBadRequest, "invalid json body")
		return
	}

	if err := action(req.Subnet); err != nil {
		s.respondWithError(w, http.StatusBadRequest, err.Error())
		return
	}

	w.WriteHeader(http.StatusOK)
}

func (s *Server) respondWithError(w http.ResponseWriter, code int, msg string) {
	s.respondWithJSON(w, code, ErrorResponse{Error: msg})
}

func (s *Server) respondWithJSON(w http.ResponseWriter, code int, payload interface{}) {
	response, _ := json.Marshal(payload)
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(code)
	_, _ = w.Write(response)
}

package subnets

import (
	"fmt"
	"net"
	"sync"
)

type Subnets struct {
	mutex  sync.RWMutex
	ipnets map[string]*net.IPNet
}

func NewSubnets() *Subnets {
	return &Subnets{
		ipnets: make(map[string]*net.IPNet),
	}
}

func (s *Subnets) Add(cidr string) error {
	_, ipNet, err := net.ParseCIDR(cidr)
	if err != nil {
		return fmt.Errorf("Invalid CIDR: %w", err)
	}

	s.mutex.Lock()
	defer s.mutex.Unlock()

	s.ipnets[ipNet.String()] = ipNet
	return nil
}

func (s *Subnets) Delete(cidr string) error {
	_, ipNet, err := net.ParseCIDR(cidr)
	if err != nil {
		return fmt.Errorf("Invalid CIDR: %w", err)
	}

	s.mutex.Lock()
	defer s.mutex.Unlock()

	delete(s.ipnets, ipNet.String())
	return nil
}

func (s *Subnets) Contains(ipStr string) bool {
	ip := net.ParseIP(ipStr)
	if ip == nil {
		return false
	}

	s.mutex.RLock()
	defer s.mutex.RUnlock()

	for _, ipnet := range s.ipnets {
		if ipnet.Contains(ip) {
			return true
		}
	}
	return false
}

func (s *Subnets) ToString() []string {
	s.mutex.RLock()
	defer s.mutex.RUnlock()

	result := make([]string, 0, len(s.ipnets))
	for _, ipnet := range s.ipnets {
		result = append(result, ipnet.String())
	}
	return result
}

# ТЗ на разработку сервиса "Анти-брутфорс"

## Общее описание
Сервис предназначен для борьбы с подбором паролей при авторизации в какой-либо системе.

Сервис вызывается перед авторизацией пользователя и может либо разрешить, либо заблокировать попытку.

Предполагается, что сервис используется только для server-server, т.е. скрыт от конечного пользователя.

## Алгоритм работы
Сервис ограничивает частоту попыток авторизации для различных комбинаций параметров, например:
* не более N = 10 попыток в минуту для данного логина.
* не более M = 100 попыток в минуту для данного пароля (защита от обратного brute-force).
* не более K = 1000 попыток в минуту для данного IP (число большое, т.к. NAT).

Для подсчета и ограничения частоты запросов, можно использовать например алгоритм leaky bucket.
Или иные аналогичные: https://en.wikipedia.org/wiki/Rate_limiting

Причем сервис будет поддерживать множество bucket-ов, по одному на каждый логин/пароль/ip.

Bucket-ы можно хранить:
* в памяти (в таком случае нужно продумать удаление неактивных bucket-ов, чтобы избежать утечек памяти).
* во внешнем хранилище (например redis или СУБД, в таком случае нужно продумать производительность).

White/black листы содержат списки адресов сетей, которые обрабатываются более простым способом:
* Если входящий IP в whitelist, то сервис безусловно разрешает авторизацию (ok=true);
* Если - в blacklist, то отклоняет (ok=false).

## Архитектура
Микросервис состоит из API, базы данных для хранения настроек и black/white списков.
Опционально - хранилище для bucket'ов. Сервис должен предоставлять GRPC **или** REST API.

## Описание методов API

### Попытка авторизации
Запрос:
* login
* password
* ip

Ответ:
* ok (true/false) - сервис должен возвращать ok=true, если считает что запрос нормальный 
и ok=false, если считает что происходит bruteforce.

### Сброс bucket
Должен очистить bucket-ы соответствующие переданным login и ip.
* login
* ip

### Добавление IP в blacklist
* подсеть (IP + маска)

### Удаление IP из blacklist
* подсеть (IP + маска)

### Добавление IP в whitelist
* подсеть (IP + маска)

### Удаление IP из whitelist
* подсеть (IP + маска)

---

- Достаточно IPv4
- Пример подсети: 192.1.1.0/25 - представляет собой адрес 192.1.1.0 с маской 255.255.255.128
- Во время работы сервиса при поступлении очередного IP мы проходимся по подсетям в черных и белых списках и вычисляем, принадлежит ли IP одной из них.

---

## Конфигурация
Основные параметры конфигурации: N, M, K - лимиты по достижению которых, сервис считает попытку брутфорсом.

## Command-Line интерфейс
Необходимо так же разработать command-line интерфейс для ручного администрирования сервиса.
Через CLI должна быть возможность вызвать сброс бакета и управлять whitelist/blacklist-ом.
CLI может работать через GRPC/HTTP интерфейс.
